# STEP 2. Create `subsystem_parser.py`

Create a new Python file named exactly:

`subsystem_parser.py`

Copy this complete code into it.

Python

```
import os
import re
import json
import tempfile

from docx import Document
from docx.text.paragraph import Paragraph
from docx.table import Table
from docx.oxml.text.paragraph import CT_P
from docx.oxml.table import CT_Tbl


def normalize_header(value):
    return re.sub(r"[\s.]+", "", value).lower()


def is_valid_filename(value):

    value = value.strip()

    if not value:
        return False

    # Ignore obvious explanatory entries.
    if value.startswith("(") or value.endswith(")"):
        return False

    if ":" in value:
        return False

    if re.search(r"\.(tst|sh)$", value, re.IGNORECASE):
        return True

    # Preserve possible extensionless procedure identifiers.
    if re.match(r"^[A-Za-z0-9_-]+$", value):
        return True

    return False


def extract_filenames(cell_text, subsystem, warnings):

    if not cell_text:
        return []

    text = cell_text.replace(",", "\n")
    text = text.replace(";", "\n")

    lines = [
        line.strip()
        for line in text.splitlines()
        if line.strip()
    ]

    reconstructed = []

    for line in lines:

        # Reconstruct known split filename patterns.
        if reconstructed and (
            reconstructed[-1].endswith("-")
            or reconstructed[-1].lower().endswith(".t")
            or reconstructed[-1].lower().endswith(".s")
        ):
            reconstructed[-1] += line

        else:
            reconstructed.append(line)

    results = []
    seen = set()

    for filename in reconstructed:

        filename = re.sub(r"\s+", "", filename)

        if not filename:
            continue

        if not is_valid_filename(filename):

            warnings.append({
                "subsystem": subsystem,
                "value": filename,
                "reason": "Invalid or suspicious filename entry"
            })

            continue

        key = filename.lower()

        if key not in seen:
            seen.add(key)
            results.append(filename)

    return results


def parse_subsystem_docx(file_path):

    if not os.path.isfile(file_path):
        raise FileNotFoundError(
            "DOCX file not found: " + file_path
        )

    doc = Document(file_path)

    subsystem_data = {}
    warnings = []

    current_subsystem = None

    # Process paragraphs and tables in document order.
    for element in doc.element.body.iterchildren():

        if isinstance(element, CT_P):

            paragraph = Paragraph(element, doc)
            text = paragraph.text.strip()

            match = re.match(
                r"^\s*Subsystem\s*:\s*(.*?)\s*$",
                text,
                re.IGNORECASE
            )

            if match:

                subsystem_name = match.group(1).strip()

                if not subsystem_name:
                    current_subsystem = None
                    warnings.append({
                        "subsystem": "",
                        "value": text,
                        "reason": "Empty subsystem heading"
                    })
                    continue

                # Case-insensitive subsystem matching.
                existing_key = None

                for key in subsystem_data:
                    if key.lower() == subsystem_name.lower():
                        existing_key = key
                        break

                if existing_key:
                    current_subsystem = existing_key
                else:
                    current_subsystem = subsystem_name
                    subsystem_data[current_subsystem] = []

        elif isinstance(element, CT_Tbl):

            table = Table(element, doc)

            if not current_subsystem:

                warnings.append({
                    "subsystem": "",
                    "value": "Table found without subsystem heading",
                    "reason": "Table cannot be assigned"
                })

                continue

            filename_column = None
            header_row = None

            # Locate Filename column dynamically.
            for row_index, row in enumerate(table.rows):

                for column_index, cell in enumerate(row.cells):

                    if normalize_header(cell.text.strip()) == "filename":

                        filename_column = column_index
                        header_row = row_index
                        break

                if filename_column is not None:
                    break

            if filename_column is None:

                warnings.append({
                    "subsystem": current_subsystem,
                    "value": "Filename column",
                    "reason": "Filename header not found"
                })

                continue

            # Extract filename values below the header.
            for row in table.rows[header_row + 1:]:

                if filename_column >= len(row.cells):
                    continue

                cell_text = row.cells[filename_column].text

                filenames = extract_filenames(
                    cell_text,
                    current_subsystem,
                    warnings
                )

                for filename in filenames:

                    existing = [
                        item.lower()
                        for item in subsystem_data[current_subsystem]
                    ]

                    if filename.lower() not in existing:
                        subsystem_data[current_subsystem].append(
                            filename
                        )

    if not subsystem_data:
        raise ValueError(
            "No subsystem data was extracted from the DOCX."
        )

    return subsystem_data, warnings


def save_subsystems_json(data, output_file):

    if not data:
        raise ValueError(
            "Cannot save empty subsystem configuration."
        )

    output_directory = os.path.dirname(
        os.path.abspath(output_file)
    )

    # Write to a temporary file first.
    temp_file = None

    try:

        with tempfile.NamedTemporaryFile(
            mode="w",
            encoding="utf-8",
            suffix=".json",
            dir=output_directory,
            delete=False
        ) as file:

            temp_file = file.name

            json.dump(
                data,
                file,
                indent=4,
                ensure_ascii=False
            )

        # Verify generated JSON.
        with open(temp_file, "r", encoding="utf-8") as file:
            verified_data = json.load(file)

        if not isinstance(verified_data, dict):
            raise ValueError("Invalid JSON structure.")

        for subsystem, filenames in verified_data.items():

            if not isinstance(subsystem, str):
                raise ValueError("Invalid subsystem name.")

            if not isinstance(filenames, list):
                raise ValueError("Invalid filename list.")

            if not all(isinstance(name, str) for name in filenames):
                raise ValueError("Invalid filename entry.")

        # Replace previous configuration only after validation.
        os.replace(temp_file, output_file)
        temp_file = None

    finally:

        if temp_file and os.path.exists(temp_file):
            os.remove(temp_file)


def process_subsystem_document(input_file, output_file):

    data, warnings = parse_subsystem_docx(input_file)

    save_subsystems_json(data, output_file)

    return data, warnings
```

This file contains all three important functions:

* `parse_subsystem_docx()` – extracts subsystem information.

* `save_subsystems_json()` – validates and saves JSON.

* `process_subsystem_document()` – connects extraction and JSON generation.

The filtering rules are a starting point: unusual extensionless entries may still need review against your actual filenames. Also, the current implementation supports DOCX only, not PDF yet.

# STEP 3. Create `test_subsystem_parser.py`

Now create another Python file:

`test_subsystem_parser.py`

Copy this code:

Python

```
from subsystem_parser import process_subsystem_document


# Input DOCX file
DOCX_FILE = "subsystem.docx"

# Output JSON file
JSON_FILE = "subsystems.json"


def main():

    print("\n===================================")
    print("   SUBSYSTEM PARSER TESTING")
    print("===================================")

    try:

        data, warnings = process_subsystem_document(
            DOCX_FILE,
            JSON_FILE
        )

        print("\n========== FINAL SUBSYSTEM DATA ==========")

        for subsystem, filenames in data.items():

            print("\nSUBSYSTEM:", subsystem)

            for filename in filenames:
                print("   ", filename)

        total_subsystems = len(data)

        total_files = sum(
            len(files)
            for files in data.values()
        )

        print("\n===================================")
        print("          FINAL SUMMARY")
        print("===================================")

        print("TOTAL SUBSYSTEMS:", total_subsystems)
        print("TOTAL FILENAMES:", total_files)
        print("TOTAL WARNINGS:", len(warnings))

        if warnings:

            print("\n========== WARNINGS ==========")

            for warning in warnings:

                print("Subsystem:", warning["subsystem"])
                print("Value:", warning["value"])
                print("Reason:", warning["reason"])
                print("--------------------------")

        print("\nJSON FILE:", JSON_FILE)
        print("\nPARSER TEST COMPLETED")

    except Exception as error:

        print("\nPARSER TEST FAILED")
        print("ERROR:", error)


if __name__ == "__main__":
    main()
```
