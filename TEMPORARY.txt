# pdf_parser.py
# Basic PDF text extraction - Python 3.6.5 compatible

import re
from PyPDF2 import PdfReader


def extract_pdf_text(pdf_path):
    reader = PdfReader(pdf_path)
    pages = []

    for page in reader.pages:
        pages.append(page.extract_text() or "")

    return "\n".join(pages)


def extract_procedures(pdf_path):
    text = extract_pdf_text(pdf_path)

    procedures = []

    # Basic reconstruction pattern.
    # Adapt these patterns to your actual PDF report format.
    date_pattern = re.compile(
        r'(\d{4}-\d{2}-\d{2}\s+\d{2}:\d{2}:\d{2}(?:\.\d+)?)'
    )

    for match in date_pattern.finditer(text):
        date_time = match.group(1)

        following_text = text[match.end():match.end() + 300]

        status_match = re.search(
            r'\b(SUCCESS|FAILURE)\b',
            following_text,
            re.IGNORECASE
        )

        if status_match:
            procedures.append({
                "date_time": date_time,
                "final_status": status_match.group(1).lower()
            })

    return procedures
```

# main.py

import os
import json

from pdf_parser import extract_procedures


def process_pdfs(input_folder, output_folder, project_name=""):

    if not os.path.exists(output_folder):
        os.makedirs(output_folder)

    output_data = {}

    for filename in os.listdir(input_folder):

        if not filename.lower().endswith(".pdf"):
            continue

        pdf_path = os.path.join(input_folder, filename)

        procedure_name = os.path.splitext(filename)[0]

        try:
            executions = extract_procedures(pdf_path)

            output_data[procedure_name] = {
                "procedures": executions
            }

        except Exception as error:
            print("Error processing {}: {}".format(
                filename, error
            ))

    output_path = os.path.join(
        output_folder,
        "{}_output.json".format(project_name or "project")
    )

    with open(output_path, "w") as output_file:
        json.dump(output_data, output_file, indent=4)

    return output_path

# dashboard.py
# Python 3.6.5 compatible

import os
import json
import re
import difflib

from datetime import datetime


BASE_DIR = os.path.dirname(os.path.abspath(__file__))
SUBSYSTEM_FILE = os.path.join(BASE_DIR, "subsystems.json")

SIMILARITY_THRESHOLD = 0.90


def normalize_name(name):

    name = os.path.basename(str(name)).strip().lower()
    name = re.sub(r"\s+", "", name)

    return name


def similarity_score(name1, name2):

    return difflib.SequenceMatcher(
        None,
        normalize_name(name1),
        normalize_name(name2)
    ).ratio()


def parse_datetime(value):

    if not value:
        return None

    formats = [
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S"
    ]

    for fmt in formats:
        try:
            return datetime.strptime(str(value).strip(), fmt)
        except (ValueError, TypeError):
            pass

    return None


def load_subsystems():

    with open(SUBSYSTEM_FILE, "r") as file:
        return json.load(file)


def filter_procedure_data(
    procedure_data,
    start_date=None,
    end_date=None
):

    executions = procedure_data.get("procedures", [])

    valid = []

    for execution in executions:

        if not isinstance(execution, dict):
            continue

        execution_date = parse_datetime(
            execution.get("date_time")
        )

        if execution_date is None:
            continue

        if start_date and execution_date.date() < start_date:
            continue

        if end_date and execution_date.date() > end_date:
            continue

        valid.append((execution_date, execution))

    if not valid:
        return None

    return max(valid, key=lambda item: item[0])[1]


def find_best_match(target_name, json_names):

    target = normalize_name(target_name)

    exact = [
        name for name in json_names
        if normalize_name(name) == target
    ]

    if len(exact) == 1:
        return exact[0], "exact"

    if len(exact) > 1:
        return None, "ambiguous"

    scored = []

    for name in json_names:

        score = similarity_score(target_name, name)

        if score >= SIMILARITY_THRESHOLD:
            scored.append((name, score))

    if not scored:
        return None, "unmatched"

    scored.sort(key=lambda item: item[1], reverse=True)

    highest = scored[0][1]

    best = [
        item for item in scored
        if abs(item[1] - highest) < 0.000001
    ]

    if len(best) != 1:
        return None, "ambiguous"

    return best[0][0], "similar"


def get_date_range(json_procedures):

    dates = []

    for data in json_procedures.values():

        for execution in data.get("procedures", []):

            if not isinstance(execution, dict):
                continue

            parsed = parse_datetime(
                execution.get("date_time")
            )

            if parsed:
                dates.append(parsed)

    if not dates:
        return "Date range unavailable"

    return "{} to {}".format(
        min(dates).strftime("%d-%m-%Y"),
        max(dates).strftime("%d-%m-%Y")
    )


def process_dashboard(
    json_path,
    project_name="",
    start_date=None,
    end_date=None
):

    with open(json_path, "r") as file:
        json_data = json.load(file)

    json_procedures = {
        name: data
        for name, data in json_data.items()
        if name != "Insights" and isinstance(data, dict)
    }

    subsystems = load_subsystems()

    json_names = list(json_procedures.keys())

    assigned_names = set()
    reserved_names = set()

    results = []

    for subsystem_name, configured_names in subsystems.items():

        success = []
        failure = []
        unmatched = []
        ambiguous = []

        for configured_name in configured_names:

            matched_name, match_type = find_best_match(
                configured_name,
                json_names
            )

            if matched_name is None:

                if match_type == "ambiguous":
                    ambiguous.append(configured_name)

                else:
                    unmatched.append(configured_name)

                continue

            if matched_name in assigned_names:
                ambiguous.append(configured_name)
                reserved_names.add(matched_name)
                continue

            assigned_names.add(matched_name)

            latest = filter_procedure_data(
                json_procedures[matched_name],
                start_date,
                end_date
            )

            if latest is None:
                unmatched.append(configured_name)
                continue

            status = str(
                latest.get("final_status", "")
            ).strip().lower()

            if status == "success":
                success.append(configured_name)

            elif status == "failure":
                failure.append(configured_name)

            else:
                unmatched.append(configured_name)

        total = len(success) + len(failure)

        results.append({
            "name": subsystem_name,
            "success": success,
            "failure": failure,
            "unmatched": unmatched,
            "ambiguous": ambiguous,
            "success_count": len(success),
            "failure_count": len(failure),
            "unmatched_count": len(unmatched),
            "ambiguous_count": len(ambiguous),
            "total": total,
            "success_percentage": round(
                len(success) * 100.0 / total, 2
            ) if total else 0,
            "failure_percentage": round(
                len(failure) * 100.0 / total, 2
            ) if total else 0
        })

    unassigned_success = []
    unassigned_failure = []

    for name, data in json_procedures.items():

        if name in assigned_names or name in reserved_names:
            continue

        latest = filter_procedure_data(
            data,
            start_date,
            end_date
        )

        if latest is None:
            continue

        status = str(
            latest.get("final_status", "")
        ).strip().lower()

        if status == "success":
            unassigned_success.append(name)

        elif status == "failure":
            unassigned_failure.append(name)

    return {
        "project_name": project_name,
        "date_range": get_date_range(json_procedures),
        "subsystems": results,
        "unassigned_procedures": {
            "success": unassigned_success,
            "failure": unassigned_failure,
            "success_count": len(unassigned_success),
            "failure_count": len(unassigned_failure),
            "total": (
                len(unassigned_success)
                + len(unassigned_failure)
            )
        }
    }
```
