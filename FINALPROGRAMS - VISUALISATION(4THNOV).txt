# dashboard.py
# Dashboard processing logic
# Compatible with Python 3.6.5

import json
import os
import re
import difflib
from datetime import datetime


BASE_DIR = os.path.dirname(os.path.abspath(__file__))
SUBSYSTEM_FILE = os.path.join(BASE_DIR, "subsystems.json")

SIMILARITY_THRESHOLD = 0.90


# --------------------------------------------------
# 1. NORMALIZE FILENAMES
# --------------------------------------------------

def normalize_name(name):
    """Normalize filenames for comparison."""

    name = os.path.basename(str(name)).strip().lower()
    name = re.sub(r"\s+", "", name)

    return name


# --------------------------------------------------
# 2. CALCULATE SIMILARITY
# --------------------------------------------------

def similarity_score(name1, name2):

    first = normalize_name(name1)
    second = normalize_name(name2)

    return difflib.SequenceMatcher(
        None, first, second
    ).ratio()


# --------------------------------------------------
# 3. PARSE DATE AND TIME
# --------------------------------------------------

def parse_datetime(value):
    """Parse execution date and time."""

    if not value:
        return None

    formats = [
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S",
    ]

    for date_format in formats:

        try:
            return datetime.strptime(
                str(value).strip(),
                date_format
            )

        except (ValueError, TypeError):
            continue

    return None


# --------------------------------------------------
# 4. LOAD SUBSYSTEM CONFIGURATION
# --------------------------------------------------

def load_subsystems():

    with open(SUBSYSTEM_FILE, "r") as file:
        return json.load(file)


# --------------------------------------------------
# 5. FIND BEST PROCEDURE MATCH
# --------------------------------------------------

def find_best_match(target_name, json_names):
    """
    Match configured filenames against JSON filenames.

    Exact matching is preferred.
    Otherwise, similarity must be at least 90%.
    Equally scoring matches are treated as ambiguous.
    """

    target = normalize_name(target_name)

    # Exact / case-insensitive matching.

    exact_matches = [
        name for name in json_names
        if normalize_name(name) == target
    ]

    if len(exact_matches) == 1:
        return exact_matches[0], "exact"

    if len(exact_matches) > 1:
        return None, "ambiguous"

    # Similarity matching.

    scored_matches = []

    for name in json_names:

        score = similarity_score(
            target_name,
            name
        )

        if score >= SIMILARITY_THRESHOLD:
            scored_matches.append((name, score))

    if not scored_matches:
        return None, "unmatched"

    scored_matches.sort(
        key=lambda item: item[1],
        reverse=True
    )

    highest_score = scored_matches[0][1]

    best_matches = [
        item for item in scored_matches
        if abs(item[1] - highest_score) < 0.000001
    ]

    if len(best_matches) != 1:
        return None, "ambiguous"

    return best_matches[0][0], "similar"


# --------------------------------------------------
# 6. GET POSSIBLE AMBIGUOUS MATCHES
# --------------------------------------------------

def get_ambiguous_candidates(target_name, json_names):
    """
    Identify JSON filenames involved in an ambiguous match.

    These candidates will not be incorrectly displayed
    as unassigned procedures.
    """

    target = normalize_name(target_name)

    exact_matches = [
        name for name in json_names
        if normalize_name(name) == target
    ]

    if len(exact_matches) > 1:
        return exact_matches

    scored_matches = []

    for name in json_names:

        score = similarity_score(
            target_name,
            name
        )

        if score >= SIMILARITY_THRESHOLD:
            scored_matches.append((name, score))

    if not scored_matches:
        return []

    highest_score = max(
        item[1] for item in scored_matches
    )

    return [
        item[0] for item in scored_matches
        if abs(item[1] - highest_score) < 0.000001
    ]


# --------------------------------------------------
# 7. FIND LATEST EXECUTION
# --------------------------------------------------

def get_latest_execution(procedure_data):
    """
    Find the latest execution using date_time.

    mode_executions is ignored.
    """

    executions = procedure_data.get("procedures", [])

    valid_executions = []

    for execution in executions:

        if not isinstance(execution, dict):
            continue

        execution_date = parse_datetime(
            execution.get("date_time", "")
        )

        if execution_date is not None:

            valid_executions.append(
                (execution_date, execution)
            )

    if not valid_executions:
        return None

    valid_executions.sort(
        key=lambda item: item[0],
        reverse=True
    )

    return valid_executions[0][1]


# --------------------------------------------------
# 8. CALCULATE OVERALL DATE RANGE
# --------------------------------------------------

def get_date_range(json_procedures):
    """
    Find the oldest and latest execution dates
    across the entire generated JSON.
    """

    all_dates = []

    for procedure_data in json_procedures.values():

        executions = procedure_data.get(
            "procedures", []
        )

        for execution in executions:

            if not isinstance(execution, dict):
                continue

            execution_date = parse_datetime(
                execution.get("date_time", "")
            )

            if execution_date is not None:
                all_dates.append(execution_date)

    if not all_dates:
        return "Date range unavailable"

    oldest_date = min(all_dates)
    latest_date = max(all_dates)

    return (
        oldest_date.strftime("%d-%m-%Y")
        + " to "
        + latest_date.strftime("%d-%m-%Y")
    )


# --------------------------------------------------
# 9. PROCESS DASHBOARD
# --------------------------------------------------

def process_dashboard(json_path, project_name=""):
    """
    Process generated JSON and prepare dashboard data.

    Returns:
        project_name
        date_range
        subsystems
        unassigned_procedures
    """

    subsystems = load_subsystems()

    with open(json_path, "r") as json_file:
        json_data = json.load(json_file)

    # Ignore global Insights.

    json_procedures = {
        name: data
        for name, data in json_data.items()
        if name != "Insights"
        and isinstance(data, dict)
    }

    json_names = list(json_procedures.keys())

    # Calculate date range across ALL executions.

    date_range = get_date_range(json_procedures)

    results = []

    # Successfully assigned JSON procedures.

    assigned_names = set()

    # JSON procedures involved in ambiguous matching.

    reserved_names = set()

    # Process every configured subsystem.

    for subsystem_name, configured_names in subsystems.items():

        success_names = []
        failure_names = []
        unmatched_names = []
        ambiguous_names = []

        for configured_name in configured_names:

            matched_name, match_type = find_best_match(
                configured_name,
                json_names
            )

            # No unique matching procedure.

            if matched_name is None:

                if match_type == "ambiguous":

                    ambiguous_names.append(configured_name)

                    candidates = get_ambiguous_candidates(
                        configured_name,
                        json_names
                    )

                    reserved_names.update(candidates)

                else:

                    unmatched_names.append(configured_name)

                continue

            # Prevent assigning the same JSON procedure twice.

            if matched_name in assigned_names:

                ambiguous_names.append(configured_name)
                reserved_names.add(matched_name)

                continue

            # Reserve the matched procedure.

            assigned_names.add(matched_name)

            # Get its latest execution.

            latest = get_latest_execution(
                json_procedures[matched_name]
            )

            if latest is None:

                unmatched_names.append(configured_name)
                continue

            # Read latest final status.

            final_status = str(
                latest.get("final_status", "")
            ).strip().lower()

            if final_status == "success":

                success_names.append(configured_name)

            elif final_status == "failure":

                failure_names.append(configured_name)

            else:

                unmatched_names.append(configured_name)

        # Calculate subsystem counts.

        success_count = len(success_names)
        failure_count = len(failure_names)

        total = success_count + failure_count

        success_percentage = (
            round(success_count * 100.0 / total, 2)
            if total else 0
        )

        failure_percentage = (
            round(failure_count * 100.0 / total, 2)
            if total else 0
        )

        results.append({

            "name": subsystem_name,

            "success": success_names,
            "failure": failure_names,

            "unmatched": unmatched_names,
            "ambiguous": ambiguous_names,

            "success_count": success_count,
            "failure_count": failure_count,

            "unmatched_count": len(unmatched_names),
            "ambiguous_count": len(ambiguous_names),

            "total": total,

            "success_percentage": success_percentage,
            "failure_percentage": failure_percentage

        })

    # --------------------------------------------------
    # 10. IDENTIFY UNASSIGNED PROCEDURES
    # --------------------------------------------------

    unassigned_success = []
    unassigned_failure = []

    for procedure_name, procedure_data in json_procedures.items():

        # Skip already assigned or ambiguous candidates.

        if procedure_name in assigned_names:
            continue

        if procedure_name in reserved_names:
            continue

        # Get latest execution.

        latest = get_latest_execution(procedure_data)

        if latest is None:
            continue

        final_status = str(
            latest.get("final_status", "")
        ).strip().lower()

        # Separate Success and Failure.

        if final_status == "success":

            unassigned_success.append(procedure_name)

        elif final_status == "failure":

            unassigned_failure.append(procedure_name)

    # --------------------------------------------------
    # 11. RETURN FINAL DASHBOARD DATA
    # --------------------------------------------------

    unassigned_procedures = {

        "success": unassigned_success,

        "failure": unassigned_failure,

        "success_count": len(unassigned_success),

        "failure_count": len(unassigned_failure),

        "total": (
            len(unassigned_success)
            + len(unassigned_failure)
        )

    }

    return {


        "project_name": project_name,

        "date_range": date_range,

        "subsystems": results,

        "unassigned_procedures": unassigned_procedures

    }

# app.py
# Flask application for Satellite Testing Dashboard
# Compatible with Python 3.6.5

import os
import glob
import subprocess
import sys

from flask import (
    Flask,
    render_template,
    request,
    flash
)

from werkzeug.utils import secure_filename

from dashboard import process_dashboard


BASE_DIR = os.path.dirname(os.path.abspath(__file__))

INPUT_DIR = os.path.join(BASE_DIR, "input")
OUTPUT_DIR = os.path.join(BASE_DIR, "output")

ALLOWED_EXTENSIONS = set(["pdf"])


app = Flask(__name__)

app.secret_key = "satellite_dashboard_local_key"


# --------------------------------------------------
# 1. CREATE REQUIRED FOLDERS
# --------------------------------------------------

def create_folders():

    for folder in [INPUT_DIR, OUTPUT_DIR]:

        if not os.path.isdir(folder):
            os.makedirs(folder)


# --------------------------------------------------
# 2. CHECK FILE EXTENSION
# --------------------------------------------------

def allowed_file(filename):

    return (
        "." in filename
        and filename.rsplit(".", 1)[1].lower()
        in ALLOWED_EXTENSIONS
    )


# --------------------------------------------------
# 3. GET LATEST GENERATED JSON
# --------------------------------------------------

def get_latest_json():

    files = glob.glob(
        os.path.join(OUTPUT_DIR, "output_*.json")
    )

    if not files:
        return None

    return max(
        files,
        key=os.path.getmtime
    )


# --------------------------------------------------
# 4. CLEAR OLD INPUT PDF FILES
# --------------------------------------------------

def clear_old_input_pdfs():

    for path in glob.glob(
        os.path.join(INPUT_DIR, "*.pdf")
    ):

        if os.path.isfile(path):
            os.remove(path)


# --------------------------------------------------
# 5. RUN EXISTING PDF CONVERTER
# --------------------------------------------------

def run_pdf_converter():

    main_path = os.path.join(
        BASE_DIR,
        "main.py"
    )

    result = subprocess.run(

        [sys.executable, main_path],

        cwd=BASE_DIR,

        stdout=subprocess.PIPE,

        stderr=subprocess.PIPE,

        universal_newlines=True
    )

    if result.returncode != 0:

        error_message = (
            result.stderr
            or result.stdout
            or "PDF conversion failed."
        )

        raise RuntimeError(error_message)

    return result.stdout


# --------------------------------------------------
# 6. MAIN DASHBOARD ROUTE
# --------------------------------------------------

@app.route("/", methods=["GET", "POST"])
def index():

    create_folders()

    dashboard_data = None
    project_name = ""

    if request.method == "POST":

        project_name = request.form.get(
            "project_name", ""
        ).strip()

        uploaded_files = request.files.getlist(
            "pdf_files"
        )

        # Validate project name.

        if not project_name:

            flash("Please enter a project name.")

        else:

            valid_files = [
                file for file in uploaded_files
                if file
                and file.filename
                and allowed_file(file.filename)
            ]

            if not valid_files:

                flash("Please select at least one valid PDF file.")

            else:

                try:

                    # Remove previous input PDFs.

                    clear_old_input_pdfs()

                    # Save uploaded PDFs.

                    saved_count = 0

                    for uploaded_file in valid_files:

                        filename = secure_filename(
                            uploaded_file.filename
                        )

                        if not filename:
                            continue

                        save_path = os.path.join(
                            INPUT_DIR,
                            filename
                        )

                        uploaded_file.save(save_path)

                        saved_count += 1

                    if saved_count == 0:

                        raise RuntimeError(
                            "No valid PDF files could be saved."
                        )

                    # Execute existing converter.

                    converter_message = run_pdf_converter()

                    # Locate generated JSON.

                    json_path = get_latest_json()

                    if json_path is None:

                        raise RuntimeError(
                            "The converter did not generate "
                            "an output JSON file."
                        )

                    # Process JSON.

                    dashboard_data = process_dashboard(
                        json_path,
                        project_name
                    )

                    flash(
                        str(saved_count)
                        + " PDF file(s) processed successfully."
                    )

                except Exception as error:

                    flash(
                        "Error: " + str(error)
                    )

    return render_template(

        "index.html",

        dashboard_data=dashboard_data,

        project_name=project_name

    )


# --------------------------------------------------
# 7. RUN APPLICATION
# --------------------------------------------------

if __name__ == "__main__":

    create_folders()

    print("=" * 50)
    print("SATELLITE TESTING DASHBOARD")
    print("=" * 50)

    print(
        "Open your browser at http://127.0.0.1:5000"
    )

    print("Press CTRL+C to stop the server.")

    app.run(

        host="127.0.0.1",

        port=5000,

        debug=False

    )

# main.py
# Main program for converting every PDF in input/ into one JSON file.
# Compatible with Python 3.6 and PyMuPDF 1.18.x.

import glob
import json
import os
import re
import sys
from collections import OrderedDict
from datetime import datetime

import fitz

from pdf_parser import parse_pdf_text, normalize_procedure_name, validate_execution


BASE_DIR = os.path.dirname(os.path.abspath(__file__))
INPUT_DIR = os.path.join(BASE_DIR, "input")
OUTPUT_DIR = os.path.join(BASE_DIR, "output")


def get_pdf_document(path):
    """Open a PDF using the PyMuPDF API available in this environment."""
    return fitz.open(path)


def get_page_count(document):
    """Support both older and newer PyMuPDF page-count properties."""
    if hasattr(document, "pageCount"):
        return document.pageCount
    return document.page_count


def get_page(document, page_number):
    """Support both older and newer PyMuPDF page-loading methods."""
    if hasattr(document, "loadPage"):
        return document.loadPage(page_number)
    return document.load_page(page_number)


def get_page_text(page):
    """Extract selectable text from one page."""
    if hasattr(page, "getText"):
        return page.getText("text")
    return page.get_text("text")


def read_pdf_text_incrementally(path):
    """
    Read one page at a time, instead of asking PyMuPDF to extract the whole
    document in one operation. The text is kept together so procedure rows
    that continue across page boundaries can still be parsed as one sequence.
    """
    document = None
    page_text_parts = []
    page_count = 0

    try:
        document = get_pdf_document(path)
        page_count = get_page_count(document)

        for page_number in range(page_count):
            page = get_page(document, page_number)
            page_text_parts.append(get_page_text(page) or "")

            # Keep a page boundary, but do not reset procedure or mode context.
            page_text_parts.append("\n")

    finally:
        if document is not None:
            document.close()

    return "".join(page_text_parts), page_count


def list_input_pdfs():
    """Return all PDF files directly inside the input folder."""
    pattern = os.path.join(INPUT_DIR, "*.pdf")
    return sorted(glob.glob(pattern), key=lambda path: os.path.basename(path).lower())


def ensure_folders_exist():
    """Check the folders that are part of the agreed project layout."""
    if not os.path.isdir(INPUT_DIR):
        raise RuntimeError(
            "The input folder was not found:\n" + INPUT_DIR
        )

    if not os.path.isdir(OUTPUT_DIR):
        os.makedirs(OUTPUT_DIR)


def add_execution(output_data, execution, pdf_name):
    """Group an execution under its procedure name."""
    procedure_name = (execution.get("Procedure Name") or "").strip()

    if not procedure_name:
        procedure_name = "UNKNOWN_PROCEDURE_" + str(len(output_data) + 1)
        print(
            "WARNING: A procedure name could not be detected in "
            + pdf_name + ". It was placed under " + procedure_name + "."
        )

    normalized_name = normalize_procedure_name(procedure_name)

    # Reuse the existing key if the same procedure name appears with different
    # capitalization or extra spaces in another execution.
    existing_key = None
    for key in output_data.keys():
        if key != "Insights" and normalize_procedure_name(key) == normalized_name:
            existing_key = key
            break

    if existing_key is None:
        existing_key = procedure_name
        output_data[existing_key] = {"procedures": []}

    clean_execution = OrderedDict()
    clean_execution["date_time"] = execution.get("date_time", "")
    clean_execution["Dur"] = execution.get("Dur", 0)
    clean_execution["Status"] = execution.get("Status", {})
    clean_execution["final_status"] = execution.get("final_status", "Success")
    clean_execution["final_status_reason"] = execution.get(
        "final_status_reason", "None"
    )
    clean_execution["mode_executions"] = execution.get("mode_executions", {})

    output_data[existing_key]["procedures"].append(clean_execution)


def remove_old_generated_outputs():
    """Remove only files matching this program's generated-output filename."""
    pattern = os.path.join(OUTPUT_DIR, "output_*.json")
    for path in glob.glob(pattern):
        if os.path.isfile(path):
            os.remove(path)


def validate_output_data(output_data):
    """Check basic JSON structure before replacing the previous output."""
    if "Insights" not in output_data:
        raise ValueError("The output is missing the global Insights object.")

    for procedure_name, procedure_group in output_data.items():
        if procedure_name == "Insights":
            continue

        if not isinstance(procedure_group, dict):
            raise ValueError("Invalid procedure group: " + str(procedure_name))

        if not isinstance(procedure_group.get("procedures"), list):
            raise ValueError(
                "Procedure group has no procedures list: " + str(procedure_name)
            )

        for execution in procedure_group["procedures"]:
            issues = validate_execution(execution)
            if issues:
                raise ValueError(
                    "Output validation failed for " + str(procedure_name) +
                    ": " + "; ".join(issues)
                )


def write_output_safely(output_data):
    """
    Write and validate a temporary JSON file first. Only after it succeeds,
    remove the previous generated JSON and rename the new file into place.
    """
    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    output_name = "output_" + timestamp + ".json"
    final_path = os.path.join(OUTPUT_DIR, output_name)
    temporary_path = final_path + ".tmp"

    try:
        with open(temporary_path, "w", encoding="utf-8") as output_file:
            json.dump(
                output_data,
                output_file,
                indent=4,
                ensure_ascii=False,
            )
            output_file.write("\n")

        # Re-open the temporary file to ensure the JSON can be read.
        with open(temporary_path, "r", encoding="utf-8") as check_file:
            json.load(check_file)

        # Preserve the previous output until the new file has been written
        # and successfully parsed as JSON.
        remove_old_generated_outputs()
        os.rename(temporary_path, final_path)

    except Exception:
        if os.path.exists(temporary_path):
            try:
                os.remove(temporary_path)
            except OSError:
                pass
        raise

    return final_path


def main():
    print("=" * 60)
    print("PDF to JSON converter")
    print("=" * 60)

    try:
        ensure_folders_exist()
    except Exception as error:
        print("ERROR:", error)
        return 1

    pdf_paths = list_input_pdfs()

    if not pdf_paths:
        print("No PDF files were found in the input folder:")
        print(INPUT_DIR)
        print("Put new_dataset.pdf (and any other PDFs) inside input/ and run again.")
        return 1

    output_data = OrderedDict()
    pdf_names = []
    total_pages_read = 0
    total_procedures = 0
    total_warnings = 0
    pdfs_read = 0

    for pdf_path in pdf_paths:
        pdf_name = os.path.basename(pdf_path)
        print("\nReading PDF:", pdf_name)

        try:
            pdf_text, page_count = read_pdf_text_incrementally(pdf_path)
            total_pages_read += page_count
            pdf_names.append(pdf_name)

            executions, warnings = parse_pdf_text(pdf_text, pdf_name)
            pdfs_read += 1

            if not executions:
                print("WARNING: No procedure executions were extracted from", pdf_name)

            for execution in executions:
                add_execution(output_data, execution, pdf_name)
                total_procedures += 1

            for warning in warnings:
                total_warnings += 1
                print("WARNING:", warning)

            print(
                "Finished:", pdf_name,
                "| Pages:", page_count,
                "| Procedure executions:", len(executions),
                "| Warnings:", len(warnings),
            )

        except Exception as error:
            print("ERROR: Could not process", pdf_name)
            print("Reason:", str(error))
            print("The other PDFs will still be processed.")

    if pdfs_read == 0:
        print("\nNo PDF was processed successfully. No new output was created.")
        return 1

    # Insights is one global object, added once at the end.
    output_data["Insights"] = OrderedDict([
        ("Total PDFs read", pdfs_read),
        ("PDF names", pdf_names),
        ("Total Pages read", total_pages_read),
    ])

    try:
        validate_output_data(output_data)
        output_path = write_output_safely(output_data)
    except Exception as error:
        print("\nERROR: The output could not be safely saved.")
        print("Reason:", str(error))
        print("Any previous generated JSON was kept if writing had not completed.")
        return 1

    print("\n" + "=" * 60)
    print("Conversion finished.")
    print("PDFs read:", pdfs_read)
    print("Total pages read:", total_pages_read)
    print("Procedure executions extracted:", total_procedures)
    print("Warnings raised:", total_warnings)
    print("Output file:", output_path)
    print("=" * 60)

    if total_warnings:
        print(
            "\nPlease review the WARNING messages above and spot-check the JSON "
            "against the PDF before using it."
        )
    else:
        print(
            "\nPlease still spot-check a few records against the source PDF "
            "before relying on the output."
        )

    return 0


if __name__ == "__main__":
    sys.exit(main())
