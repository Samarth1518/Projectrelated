def save_step_status(step_id, step_lines, status_data):
    # Join all lines belonging to this STEP
    step_text = " ".join(step_lines)
    # Look for a valid status
    status_match = re.search(
        r"\b(Success|Failure|Suspend|Progress)\b",
        step_text,
        re.IGNORECASE
    )
    if status_match:
        status = status_match.group(1).lower()
        if status not in status_data:
            status_data[status] = []
        status_data[status].append(step_id)
def extract_status(text):
    status_data = {}
    lines = text.splitlines()
    current_step_id = None
    current_step_lines = []
    for line in lines:
        line = line.strip()
        # Ignore empty lines
        if not line:
            continue
        # Check for a genuine STEP ID
        step_match = re.match(r"^(\d+\.\d+)\b", line)
        if step_match:
            # Finish previous STEP
            if current_step_id is not None:
                save_step_status(
                    current_step_id,
                    current_step_lines,
                    status_data
                )
            # Store new STEP ID
            current_step_id = step_match.group(1)
            current_step_lines = []
            # Get remaining text after STEP ID
            remaining_text = line[step_match.end():].strip()
            if remaining_text:
                current_step_lines.append(remaining_text)
        else:
            # Add line to current STEP
            if current_step_id is not None:
                current_step_lines.append(line)
    # Finish final STEP
    if current_step_id is not None:
        save_step_status(
            current_step_id,
            current_step_lines,
            status_data
        )
    return status_data

10. Critical limitation
This code is a structural starting point. It supports all four status words, but it still searches the collected STEP
text for the first matching status word. Therefore it is NOT yet the final protection against status words inside the
STEP description.
01.0001
Check Success condition
11:44:20
Failure
The required result here is failure fi 01.0001. The intermediate code could see Success first. Therefore, do not
consider this extraction part finished until the actual college PDF layout has been checked.
