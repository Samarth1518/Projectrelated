def validate_dates(start_date, end_date):
    """
    Validate the selected date range.

    Expected format: YYYY-MM-DD
    """

    try:

        start = datetime.strptime(
            start_date,
            "%Y-%m-%d"

            end_date,

    except (ValueError, TypeError):

def save_uploaded_pdfs(uploaded_files):
    """
    Save uploaded PDFs into the input folder.
    """

    saved_count = 0

    for uploaded_file in uploaded_files:

        filename = secure_filename(
            uploaded_file.filename

        if not filename:
            continue

        save_path = os.path.join(
            INPUT_DIR,
            filename
        )

        uploaded_file.save(save_path)

        saved_count += 1

    if saved_count == 0:

        raise RuntimeError(
            "No valid PDF files could be saved."
        )

    return saved_count

def fetch_subsystems_from_api(project_name):
    """
    Call the subsystem API using the entered project name.

    Expected API response:

    {
        "data": {
            "POWER": [
                "ab-abc.tst",
                "bc-bst.tst"
            ],
            "ILC": [
                "ab-sdc.tst",
                "rc-bst.tst"
            ]
        }
    }

    The subsystem names returned by the API become
    the keys in subsystems.json.
    """

    try:

        response = requests.post(

            API_URL,

            json={
                "project_name": project_name
            },

            headers={
                "Content-Type": "application/json",
                "Accept": "application/json"
            },

            timeout=API_TIMEOUT

        )

        # Check HTTP response status.

        response.raise_for_status()

    except requests.exceptions.Timeout:

        raise RuntimeError(
            "Subsystem API request timed out. "
            "Please try again."
        )

    except requests.exceptions.ConnectionError:

        raise RuntimeError(
            "Could not connect to the subsystem API. "
            "Please check the API server and try again."
        )

    except requests.exceptions.HTTPError:

        raise RuntimeError(
            "Subsystem API returned HTTP error "
            + str(response.status_code)
            + ". Please try again."
        )

    except requests.exceptions.RequestException as error:

        raise RuntimeError(
            "Subsystem API request failed: "
            + str(error)
        )

    # Convert API response into Python dictionary.

    try:

        api_response = response.json()

    except ValueError:

        raise RuntimeError(
            "Subsystem API returned an invalid JSON response."
        )

    # Validate the outer response structure.

    if not isinstance(api_response, dict):

        raise RuntimeError(
            "Invalid subsystem API response format."
        )

    subsystem_data = api_response.get("data")

    if not isinstance(subsystem_data, dict):

        raise RuntimeError(
            "Subsystem API response does not contain "
            "the expected 'data' object."
        )

    if not subsystem_data:

        raise RuntimeError(
            "The API returned no subsystems for project '"
            + project_name + "'."
        )

    # Validate and clean the subsystem information.

    clean_subsystems = {}

    for subsystem_name, filenames in subsystem_data.items():

        if not isinstance(subsystem_name, str):

            raise RuntimeError(
                "API returned an invalid subsystem name."
            )

        subsystem_name = subsystem_name.strip()

        if not subsystem_name:

            raise RuntimeError(
                "API returned an empty subsystem name."
            )

        if subsystem_name in clean_subsystems:

            raise RuntimeError(
                "API returned duplicate subsystem names."
            )

        if not isinstance(filenames, list):

            raise RuntimeError(
                "Invalid filename list for subsystem: "
                + subsystem_name
            )

        clean_filenames = []

        for filename in filenames:

            if not isinstance(filename, str):

                raise RuntimeError(
                    "Invalid filename received for subsystem: "
                    + subsystem_name
                )

            filename = filename.strip()

            if not filename:

                raise RuntimeError(
                    "An empty filename was received for subsystem: "
                    + subsystem_name
                )

            clean_filenames.append(filename)

        clean_subsystems[subsystem_name] = clean_filenames

    return clean_subsystems

def save_subsystems_safely(subsystem_data):
    """
    Save the new subsystem configuration safely.

    The previous subsystems.json is preserved if writing
    or JSON validation fails.
    """

    temporary_file = SUBSYSTEM_FILE + ".tmp"

    try:

        # Write new data into a temporary file first.

        with open(
            temporary_file,
            "w",
            encoding="utf-8"
        ) as json_file:

            json.dump(
                subsystem_data,
                json_file,
                indent=4,
                ensure_ascii=False
            )

            json_file.write("\n")

        # Verify that the temporary JSON is valid.

        with open(
            temporary_file,
            "r",
            encoding="utf-8"
        ) as json_file:

            verified_data = json.load(json_file)

        if verified_data != subsystem_data:

            raise RuntimeError(
                "Generated subsystem JSON verification failed."
            )

        # Replace the old file only after successful validation.

        os.replace(
            temporary_file,
            SUBSYSTEM_FILE
        )

    except Exception:

        if os.path.exists(temporary_file):

            try:
                os.remove(temporary_file)

            except OSError:
                pass

        raise

# 11. RUN EXISTING PDF CONVERTER

def run_pdf_converter():
    """
    Execute main.py.

    main.py generates output_YYYY-MM-DD_HH-MM-SS.json.
    """

    main_path = os.path.join(
        BASE_DIR,
        "main.py"
    )

    result = __import__("subprocess").run(

        [sys.executable, main_path],

        cwd=BASE_DIR,

        stdout=__import__("subprocess").PIPE,

        stderr=__import__("subprocess").PIPE,

        universal_newlines=True
    )

    if result.returncode != 0:

        error_message = (
            result.stderr
            or result.stdout
            or "PDF conversion failed."
        )

        raise RuntimeError(
            "PDF processing failed:\n" + error_message
        )

    # main.py prints the generated output path.
    # Read that exact path instead of accidentally using
    # an older output JSON.

    output_path = None

    for line in result.stdout.splitlines():

        if line.strip().startswith("Output file:"):

            output_path = line.split(
                "Output file:",
                1
            )[1].strip()

            break

    if not output_path:

        raise RuntimeError(
            "PDF converter finished but did not report "
            "the generated output JSON path."
        )

    if not os.path.isfile(output_path):

        raise RuntimeError(
            "The PDF converter did not generate "
            "the expected output JSON file."
        )

    return output_path

```
# Compatible with Python 3.6.5

import json
import os
import re
import difflib

from datetime import datetime, date


BASE_DIR = os.path.dirname(os.path.abspath(__file__))
SUBSYSTEM_FILE = os.path.join(BASE_DIR, "subsystems.json")

SIMILARITY_THRESHOLD = 0.90


# --------------------------------------------------
# 1. NORMALIZE FILENAMES
# --------------------------------------------------

def normalize_name(name):
    """Normalize filenames for comparison."""

    name = os.path.basename(str(name)).strip().lower()
    name = re.sub(r"\s+", "", name)

    return name


# --------------------------------------------------
# 2. CALCULATE SIMILARITY
# --------------------------------------------------

def similarity_score(name1, name2):

    first = normalize_name(name1)
    second = normalize_name(name2)

    return difflib.SequenceMatcher(
        None, first, second
    ).ratio()


# --------------------------------------------------
# 3. PARSE DATE AND TIME
# --------------------------------------------------

def parse_datetime(value):
    """Parse supported execution date/time formats."""

    if not value:
        return None

    formats = [
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S",
    ]

    for date_format in formats:

        try:
            return datetime.strptime(
                str(value).strip(),
                date_format
            )

        except (ValueError, TypeError):
            continue

    return None


# --------------------------------------------------
# 4. PARSE USER DATE INPUT
# --------------------------------------------------

def parse_filter_date(value, field_name):

    if value is None or str(value).strip() == "":
        return None

    try:
        return datetime.strptime(
            str(value).strip(),
            "%Y-%m-%d"
        ).date()

    except ValueError:

        raise ValueError(
            "Invalid " + field_name +
            ". Please select a valid date."
        )


# --------------------------------------------------
# 5. LOAD SUBSYSTEM CONFIGURATION
# --------------------------------------------------

def load_subsystems():

    if not os.path.isfile(SUBSYSTEM_FILE):

        raise RuntimeError(
            "subsystems.json was not found."
        )

    with open(SUBSYSTEM_FILE, "r") as file:

        subsystems = json.load(file)

    if not isinstance(subsystems, dict):

        raise ValueError(
            "subsystems.json must contain a JSON object."
        )

    if not subsystems:

        raise ValueError(
            "subsystems.json does not contain any subsystems."
        )

    for subsystem_name, configured_names in subsystems.items():

        if not isinstance(subsystem_name, str):

            raise ValueError(
                "Invalid subsystem name in subsystems.json."
            )

        if not isinstance(configured_names, list):

            raise ValueError(
                "Invalid procedure list for subsystem: "
                + subsystem_name
            )

        for procedure_name in configured_names:

            if not isinstance(procedure_name, str):

                raise ValueError(
                    "Invalid procedure name in subsystem: "
                    + subsystem_name
                )

    return subsystems


# --------------------------------------------------
# 6. FIND BEST PROCEDURE MATCH
# --------------------------------------------------

def find_best_match(target_name, json_names):
    """
    Exact matching is preferred.

    Otherwise, similarity must be at least 90%.

    Equally scoring matches are ambiguous.
    """

    target = normalize_name(target_name)

    exact_matches = [
        name for name in json_names
        if normalize_name(name) == target
    ]

    if len(exact_matches) == 1:

        return exact_matches[0], "exact"

    if len(exact_matches) > 1:

        return None, "ambiguous"

    scored_matches = []

    for name in json_names:

        score = similarity_score(
            target_name,
            name
        )

        if score >= SIMILARITY_THRESHOLD:

            scored_matches.append((name, score))

    if not scored_matches:

        return None, "unmatched"

    scored_matches.sort(
        key=lambda item: item[1],
        reverse=True
    )

    highest_score = scored_matches[0][1]

    best_matches = [
        item for item in scored_matches
        if abs(item[1] - highest_score) < 0.000001
    ]

    if len(best_matches) != 1:

        return None, "ambiguous"

    return best_matches[0][0], "similar"


# --------------------------------------------------
# 7. GET POSSIBLE AMBIGUOUS MATCHES
# --------------------------------------------------

def get_ambiguous_candidates(target_name, json_names):

    target = normalize_name(target_name)

    exact_matches = [
        name for name in json_names
        if normalize_name(name) == target
    ]

    if len(exact_matches) > 1:

        return exact_matches

    scored_matches = []

    for name in json_names:

        score = similarity_score(
            target_name,
            name
        )

        if score >= SIMILARITY_THRESHOLD:

            scored_matches.append((name, score))

    if not scored_matches:

        return []

    highest_score = max(
        item[1] for item in scored_matches
    )

    return [
        item[0] for item in scored_matches
        if abs(item[1] - highest_score) < 0.000001
    ]


# --------------------------------------------------
# 8. FILTER EXECUTIONS BY DATE
# --------------------------------------------------

def filter_procedure_executions(
    procedure_data,
    start_date=None,
    end_date=None
):
    """
    Return executions within the selected date range.

    Both start and end dates are inclusive.

    Invalid or missing execution timestamps are excluded
    when date filtering is enabled.
    """

    executions = procedure_data.get(
        "procedures", []
    )

    if not isinstance(executions, list):

        raise ValueError(
            "Invalid procedures list in generated JSON."
        )

    filtered = []

    for execution in executions:

        if not isinstance(execution, dict):
            continue

        execution_date = parse_datetime(
            execution.get("date_time", "")
        )

        if execution_date is None:

            if start_date is None and end_date is None:
                continue

            continue

        execution_day = execution_date.date()

        if start_date is not None:

            if execution_day < start_date:
                continue

        if end_date is not None:

            if execution_day > end_date:
                continue

        filtered.append(
            (execution_date, execution)
        )

    return filtered


# --------------------------------------------------
# 9. FIND LATEST EXECUTION
# --------------------------------------------------

def get_latest_execution(
    procedure_data,
    start_date=None,
    end_date=None
):
    """
    Find the latest execution within the selected date range.

    mode_executions is ignored.
    """

    valid_executions = filter_procedure_executions(
        procedure_data,
        start_date,
        end_date
    )

    if not valid_executions:

        return None

    valid_executions.sort(
        key=lambda item: item[0],
        reverse=True
    )

    return valid_executions[0][1]


# --------------------------------------------------
# 10. CALCULATE DATE RANGE
# --------------------------------------------------

def get_date_range(
    json_procedures,
    start_date=None,
    end_date=None
):
    """
    Calculate the actual execution date range
    within the selected date range.
    """

    all_dates = []

    for procedure_data in json_procedures.values():

        valid_executions = filter_procedure_executions(
            procedure_data,
            start_date,
            end_date
        )

        for execution_date, execution in valid_executions:

            all_dates.append(execution_date)

    if not all_dates:

        return "Date range unavailable"

    oldest_date = min(all_dates)
    latest_date = max(all_dates)

    return (
        oldest_date.strftime("%d-%m-%Y")
        + " to "
        + latest_date.strftime("%d-%m-%Y")
    )


# --------------------------------------------------
# 11. PROCESS DASHBOARD
# --------------------------------------------------

def process_dashboard(
    json_path,
    project_name="",
    start_date="",
    end_date=""
):
    """
    Process generated JSON and prepare dashboard data.

    Supports optional date filtering.

    Returns:
        project_name
        date_range
        subsystems
        unassigned_procedures
        no_executions_found
        selected_start_date
        selected_end_date
    """

    # Parse selected dates.

    start_filter = parse_filter_date(
        start_date,
        "start date"
    )

    end_filter = parse_filter_date(
        end_date,
        "end date"
    )

    if (
        start_filter is not None
        and end_filter is not None
        and start_filter > end_filter
    ):

        raise ValueError(
            "Start date cannot be later than end date."
        )

    # Load subsystem configuration.

    subsystems = load_subsystems()

    # Validate generated JSON file.

    if not os.path.isfile(json_path):

        raise RuntimeError(
            "Generated JSON file was not found."
        )

    with open(json_path, "r") as json_file:

        json_data = json.load(json_file)

    if not isinstance(json_data, dict):

        raise ValueError(
            "Generated JSON must contain a JSON object."
        )

    # Ignore global Insights.

    json_procedures = {

        name: data

        for name, data in json_data.items()

        if name != "Insights"
        and isinstance(data, dict)

    }

    json_names = list(json_procedures.keys())

    # Calculate selected date range.

    date_range = get_date_range(
        json_procedures,
        start_filter,
        end_filter
    )

    # Check whether any execution exists in selected dates.

    has_valid_execution = False

    for procedure_data in json_procedures.values():

        executions = filter_procedure_executions(
            procedure_data,
            start_filter,
            end_filter
        )

        if executions:

            has_valid_execution = True
            break

    # If no executions exist, return an explicit flag.

    if not has_valid_execution:

        return {

            "project_name": project_name,

            "date_range": date_range,

            "subsystems": [],

            "unassigned_procedures": {

                "success": [],
                "failure": [],

                "success_count": 0,
                "failure_count": 0,

                "total": 0

            },

            "no_executions_found": True,

            "selected_start_date": (
                start_filter.isoformat()
                if start_filter else ""
            ),

            "selected_end_date": (
                end_filter.isoformat()
                if end_filter else ""
            )

        }

    results = []

    # Successfully assigned JSON procedures.

    assigned_names = set()

    # JSON procedures involved in ambiguous matching.

    reserved_names = set()

    # --------------------------------------------------
    # PROCESS EACH SUBSYSTEM
    # --------------------------------------------------

    for subsystem_name, configured_names in subsystems.items():

        success_names = []
        failure_names = []
        unmatched_names = []
        ambiguous_names = []

        for configured_name in configured_names:

            matched_name, match_type = find_best_match(
                configured_name,
                json_names
            )

            # No unique matching procedure.

            if matched_name is None:

                if match_type == "ambiguous":

                    ambiguous_names.append(
                        configured_name
                    )

                    candidates = get_ambiguous_candidates(
                        configured_name,
                        json_names
                    )

                    reserved_names.update(candidates)

                else:

                    unmatched_names.append(
                        configured_name
                    )

                continue

            # Prevent duplicate assignment.

            if matched_name in assigned_names:

                ambiguous_names.append(
                    configured_name
                )

                reserved_names.add(
                    matched_name
                )

                continue

            # Get latest execution within selected dates.

            latest = get_latest_execution(
                json_procedures[matched_name],
                start_filter,
                end_filter
            )

            # No execution in selected dates.

            if latest is None:

                unmatched_names.append(
                    configured_name
                )

                continue

            # Reserve the matched procedure only after
            # confirming that a valid execution exists.

            assigned_names.add(
                matched_name
            )

            # Read latest final status.

            final_status = str(
                latest.get("final_status", "")
            ).strip().lower()

            if final_status == "success":

                success_names.append(
                    configured_name
                )

            elif final_status == "failure":

                failure_names.append(
                    configured_name
                )

            else:

                unmatched_names.append(
                    configured_name
                )

        # Calculate subsystem counts.

        success_count = len(success_names)
        failure_count = len(failure_names)

        total = success_count + failure_count

        success_percentage = (
            round(
                success_count * 100.0 / total,
                2
            )
            if total else 0
        )

        failure_percentage = (
            round(
                failure_count * 100.0 / total,
                2
            )
            if total else 0
        )

        results.append({

            "name": subsystem_name,

            "success": success_names,
            "failure": failure_names,

            "unmatched": unmatched_names,
            "ambiguous": ambiguous_names,

            "success_count": success_count,
            "failure_count": failure_count,

            "unmatched_count": len(unmatched_names),
            "ambiguous_count": len(ambiguous_names),

            "total": total,

            "success_percentage": success_percentage,
            "failure_percentage": failure_percentage

        })

    # --------------------------------------------------
    # 12. IDENTIFY UNASSIGNED PROCEDURES
    # --------------------------------------------------

    unassigned_success = []
    unassigned_failure = []

    for procedure_name, procedure_data in json_procedures.items():

        # Skip already assigned or ambiguous candidates.

        if procedure_name in assigned_names:
            continue

        if procedure_name in reserved_names:
            continue

        # Get latest execution within selected dates.

        latest = get_latest_execution(
            procedure_data,
            start_filter,
            end_filter
        )

        if latest is None:
            continue

        final_status = str(
            latest.get("final_status", "")
        ).strip().lower()

        if final_status == "success":

            unassigned_success.append(
                procedure_name
            )

        elif final_status == "failure":

            unassigned_failure.append(
                procedure_name
            )

    # --------------------------------------------------
    # 13. RETURN FINAL DASHBOARD DATA
    # --------------------------------------------------

    unassigned_procedures = {

        "success": unassigned_success,

        "failure": unassigned_failure,

        "success_count": len(unassigned_success),

        "failure_count": len(unassigned_failure),

        "total": (
            len(unassigned_success)
            + len(unassigned_failure)
        )

    }

    return {

        "project_name": project_name,

        "date_range": date_range,

        "subsystems": results,

        "unassigned_procedures": unassigned_procedures,

        "no_executions_found": False,

        "selected_start_date": (
            start_filter.isoformat()
            if start_filter else ""
        ),

        "selected_end_date": (
            end_filter.isoformat()
            if end_filter else ""
        )

    }
```

## What this version does differently

### 1. Date filtering

For example, suppose a procedure has these executions:

| Execution date | Status  |
| -------------- | ------- |
| 01-01-2026     | Success |
| 10-02-2026     | Failure |
| 15-03-2026     | Success |

If the user selects 01-02-2026 to 28-02-2026, only the February execution is considered.

The latest execution within the selected period determines the displayed status.

### 2. No executions in the selected period

If there are no valid executions in the selected date range, the function returns:

Python

```
"no_executions_found": True
```

This allows the Flask application and HTML template to display an appropriate message instead of presenting misleading subsystem statistics.

### 3. Important correction to the matching logic

A procedure is now added to `assigned_names` only after an execution is found within the selected date range.

This prevents a procedure with no execution in the selected period from being incorrectly hidden from the unassigned-procedure calculation.

```
def filter_procedure_executions(
    procedure_data,
    start_date=None,
    end_date=None
):
    """
    Return executions within the selected date range.

    Both dates are inclusive.

    When no date filter is selected, retain all executions.
    When date filtering is enabled, invalid timestamps
    are excluded.
    """

    executions = procedure_data.get(
        "procedures", []
    )

    if not isinstance(executions, list):
        raise ValueError(
            "Invalid procedures list in generated JSON."
        )

    filtered = []

    for execution in executions:

        if not isinstance(execution, dict):
            continue

        execution_date = parse_datetime(
            execution.get("date_time", "")
        )

        if execution_date is None:

            if start_date is None and end_date is None:
                filtered.append(
                    (None, execution)
                )

            continue

        execution_day = execution_date.date()

        if start_date is not None:
            if execution_day < start_date:
                continue

        if end_date is not None:
            if execution_day > end_date:
                continue

        filtered.append(
            (execution_date, execution)
        )

    return filtered
```

```
def validate_dates(start_date, end_date):

    start = None
    end = None

    if start_date:

            start = datetime.strptime(
                start_date,
                "%Y-%m-%d"
            )

        except ValueError:

    if end_date:

            end = datetime.strptime(
```
