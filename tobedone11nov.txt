# 2. `python_docx_parser.py`

New file

This file reads the uploaded Word document and extracts:

* Subsystem names.

* Filenames from the third column of each subsystem table.

* Only one occurrence of each filename within a subsystem.

* Subsystem names as the JSON keys.

It generates `subsystems.json` automatically.

Python

```
# python_docx_parser.py
# Extract subsystem names and filenames from a Word document.
# Compatible with Python 3.6.5 and python-docx.

import os
import re
import json

from docx import Document
from docx.table import Table
from docx.text.paragraph import Paragraph


BASE_DIR = os.path.dirname(os.path.abspath(__file__))

SUBSYSTEM_FILE = os.path.join(
    BASE_DIR,
    "subsystems.json"
)


def clean_text(value):
    """Clean text extracted from Word."""

    value = str(value or "").strip()
    value = re.sub(r"\s+", " ", value)

    return value


def clean_filename(value):
    """Extract a filename from a table cell."""

    value = clean_text(value)

    value = re.sub(
        r"^\s*[\d.)\-•]+\s*",
        "",
        value
    )

    value = value.strip('"\' ')

    return value


def get_cell_text(cell):
    """Read all text from a table cell."""

    parts = []

    for paragraph in cell.paragraphs:

        text = clean_text(paragraph.text)

        if text:
            parts.append(text)

    return " ".join(parts)


def get_document_elements(document):
    """
    Read paragraphs and tables in their original document order.
    """

    body = document.element.body

    for element in body.iterchildren():

        if element.tag.endswith("}p"):

            yield Paragraph(element, document)

        elif element.tag.endswith("}tbl"):

            yield Table(element, document)


def extract_subsystems(docx_path):
    """
    Extract subsystem names and filenames from tables.

    Expected layout:

        Subsystem Name

        Table:
        Column 1 | Column 2 | Filename
    """

    if not os.path.isfile(docx_path):

        raise FileNotFoundError(
            "Word document not found: " + docx_path
        )

    document = Document(docx_path)

    subsystems = {}

    current_subsystem = None

    for element in get_document_elements(document):

        # Read paragraphs before tables.

        if isinstance(element, Paragraph):

            text = clean_text(element.text)

            if text:
                current_subsystem = text

            continue

        # Process tables.

        if isinstance(element, Table):

            if not current_subsystem:
                continue

            if current_subsystem not in subsystems:

                subsystems[current_subsystem] = []

            filenames = subsystems[current_subsystem]

            # Third column = index 2.

            for row in element.rows:

                if len(row.cells) < 3:
                    continue

                filename = clean_filename(
                    get_cell_text(row.cells[2])
                )

                if not filename:
                    continue

                # Ignore table headings.

                if filename.lower() in [
                    "filename",
                    "file name",
                    "file names",
                    "procedure name",
                    "procedure filename"
                ]:

                    continue

                # Keep one occurrence within each subsystem.

                if filename not in filenames:

                    filenames.append(filename)

    # Remove empty subsystem entries.

    subsystems = {
        name: filenames
        for name, filenames in subsystems.items()
        if filenames
    }

    if not subsystems:

        raise ValueError(
            "No subsystem tables or filenames were detected. "
            "Check that subsystem names appear above their tables "
            "and filenames are in the third column."
        )

    return subsystems


def save_subsystems(subsystems):

    temporary_file = SUBSYSTEM_FILE + ".tmp"

    try:

        with open(
            temporary_file,
            "w",
            encoding="utf-8"
        ) as file:

            json.dump(
                subsystems,
                file,
                indent=4,
                ensure_ascii=False
            )

            file.write("\n")

        os.rename(
            temporary_file,
            SUBSYSTEM_FILE
        )

    except Exception:

        if os.path.exists(temporary_file):
            os.remove(temporary_file)

        raise

    return SUBSYSTEM_FILE


def process_word_document(docx_path):

    subsystems = extract_subsystems(docx_path)

    output_path = save_subsystems(subsystems)

    return subsystems, output_path
```

Important Word document format

Your Word document should follow this arrangement:

```
Power Subsystem

+-----------+-----------+----------------------+
| Column 1  | Column 2  | Column 3             |
+-----------+-----------+----------------------+
| ...       | ...       | example.tst          |
| ...       | ...       | power_check.tst      |
| ...       | ...       | power_start.tst      |
+-----------+-----------+----------------------+

Thermal Subsystem

+-----------+-----------+----------------------+
| Column 1  | Column 2  | Column 3             |
+-----------+-----------+----------------------+
| ...       | ...       | thermal_check.tst    |
| ...       | ...       | cooling_test.tst     |
+-----------+-----------+----------------------+
```

The parser reads the third column only.

```
# dashboard.py
# Dashboard processing logic.
# Compatible with Python 3.6.5.

import json
import os
import re
import difflib

from datetime import datetime


BASE_DIR = os.path.dirname(os.path.abspath(__file__))

SIMILARITY_THRESHOLD = 0.90


# --------------------------------------------------
# 1. NORMALIZE FILENAMES
# --------------------------------------------------

def normalize_name(name):

    name = os.path.basename(str(name)).strip().lower()

    # Remove file extensions.

    name = os.path.splitext(name)[0]

    # Remove spaces.

    name = re.sub(r"\s+", "", name)

    return name


# --------------------------------------------------
# 2. SIMILARITY
# --------------------------------------------------

def similarity_score(name1, name2):

    return difflib.SequenceMatcher(
        None,
        normalize_name(name1),
        normalize_name(name2)
    ).ratio()


# --------------------------------------------------
# 3. PARSE DATETIME
# --------------------------------------------------

def parse_datetime(value):

    if not value:
        return None

    formats = [
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S"
    ]

    for date_format in formats:

        try:

            return datetime.strptime(
                str(value).strip(),
                date_format
            )

        except (ValueError, TypeError):

            continue

    return None


# --------------------------------------------------
# 4. FIND BEST MATCH
# --------------------------------------------------

def find_best_match(target_name, json_names):

    target = normalize_name(target_name)

    exact_matches = [
        name for name in json_names
        if normalize_name(name) == target
    ]

    if len(exact_matches) == 1:

        return exact_matches[0], "exact"

    if len(exact_matches) > 1:

        return None, "ambiguous"

    scored_matches = []

    for name in json_names:

        score = similarity_score(
            target_name,
            name
        )

        if score >= SIMILARITY_THRESHOLD:

            scored_matches.append((name, score))

    if not scored_matches:

        return None, "unmatched"

    highest_score = max(
        item[1] for item in scored_matches
    )

    best_matches = [
        item[0]
        for item in scored_matches
        if abs(item[1] - highest_score) < 0.000001
    ]

    if len(best_matches) != 1:

        return None, "ambiguous"

    return best_matches[0], "similar"


# --------------------------------------------------
# 5. GET LATEST EXECUTION
# --------------------------------------------------

def get_latest_execution(procedure_data, start_date=None,
                         end_date=None):

    executions = procedure_data.get(
        "procedures", []
    )

    valid_executions = []

    for execution in executions:

        if not isinstance(execution, dict):
            continue

        execution_date = parse_datetime(
            execution.get("date_time", "")
        )

        if execution_date is None:
            continue

        # Apply date filtering BEFORE selecting latest.

        if start_date and execution_date < start_date:
            continue

        if end_date and execution_date > end_date:
            continue

        valid_executions.append(
            (execution_date, execution)
        )

    if not valid_executions:

        return None

    valid_executions.sort(
        key=lambda item: item[0],
        reverse=True
    )

    return valid_executions[0][1]


# --------------------------------------------------
# 6. GET DATE RANGE
# --------------------------------------------------

def get_date_range(json_procedures, start_date=None,
                   end_date=None):

    all_dates = []

    for procedure_data in json_procedures.values():

        executions = procedure_data.get(
            "procedures", []
        )

        for execution in executions:

            if not isinstance(execution, dict):
                continue

            execution_date = parse_datetime(
                execution.get("date_time", "")
            )

            if execution_date is None:
                continue

            if start_date and execution_date < start_date:
                continue

            if end_date and execution_date > end_date:
                continue

            all_dates.append(execution_date)

    if not all_dates:

        return "Date range unavailable"

    return (
        min(all_dates).strftime("%d-%m-%Y")
        + " to "
        + max(all_dates).strftime("%d-%m-%Y")
    )


# --------------------------------------------------
# 7. PROCESS DASHBOARD
# --------------------------------------------------

def process_dashboard(
    json_path,
    subsystems,
    project_name="",
    date_mode="overall",
    start_date=None,
    end_date=None
):

    with open(
        json_path,
        "r",
        encoding="utf-8"
    ) as file:

        json_data = json.load(file)

    if not isinstance(json_data, dict):

        raise ValueError("Invalid JSON structure.")

    json_procedures = {
        name: data
        for name, data in json_data.items()
        if name != "Insights"
        and isinstance(data, dict)
    }

    json_names = list(json_procedures.keys())

    # Date range.

    date_range = get_date_range(
        json_procedures,
        start_date,
        end_date
    )

    results = []

    assigned_names = set()
    reserved_names = set()

    # --------------------------------------------------
    # PROCESS EVERY SUBSYSTEM
    # --------------------------------------------------

    for subsystem_name, configured_names in subsystems.items():

        success_names = []
        failure_names = []
        unmatched_names = []
        ambiguous_names = []

        # Remove duplicate filenames within a subsystem.

        configured_names = list(
            dict.fromkeys(configured_names)
        )

        for configured_name in configured_names:

            matched_name, match_type = find_best_match(
                configured_name,
                json_names
            )

            if matched_name is None:

                if match_type == "ambiguous":

                    ambiguous_names.append(configured_name)

                    target = normalize_name(configured_name)

                    candidates = []

                    for name in json_names:

                        score = similarity_score(
                            configured_name,
                            name
                        )

                        if score >= SIMILARITY_THRESHOLD:
                            candidates.append((name, score))

                    if candidates:

                        highest = max(
                            item[1] for item in candidates
                        )

                        for name, score in candidates:

                            if abs(score - highest) < 0.000001:
                                reserved_names.add(name)

                else:

                    unmatched_names.append(configured_name)

                continue

            latest = get_latest_execution(
                json_procedures[matched_name],
                start_date,
                end_date
            )

            # No execution in selected date range.

            if latest is None:

                unmatched_names.append(configured_name)
                continue

            final_status = str(
                latest.get("final_status", "")
            ).strip().lower()

            # Count each JSON procedure only once per subsystem.

            assigned_names.add(matched_name)

            if final_status == "success":

                success_names.append(configured_name)

            elif final_status == "failure":

                failure_names.append(configured_name)

            else:

                unmatched_names.append(configured_name)

        success_count = len(success_names)
        failure_count = len(failure_names)

        total = success_count + failure_count

        success_percentage = (
            round(success_count * 100.0 / total, 2)
            if total else 0
        )

        failure_percentage = (
            round(failure_count * 100.0 / total, 2)
            if total else 0
        )

        results.append({

            "name": subsystem_name,

            "success": success_names,
            "failure": failure_names,

            "unmatched": unmatched_names,
            "ambiguous": ambiguous_names,

            "success_count": success_count,
            "failure_count": failure_count,

            "unmatched_count": len(unmatched_names),
            "ambiguous_count": len(ambiguous_names),

            "total": total,

            "success_percentage": success_percentage,
            "failure_percentage": failure_percentage

        })

    # --------------------------------------------------
    # 8. UNASSIGNED PROCEDURES
    # --------------------------------------------------

    unassigned_success = []
    unassigned_failure = []

    for procedure_name, procedure_data in json_procedures.items():

        if procedure_name in assigned_names:
            continue

        if procedure_name in reserved_names:
            continue

        latest = get_latest_execution(
            procedure_data,
            start_date,
            end_date
        )

        if latest is None:
            continue

        final_status = str(
            latest.get("final_status", "")
        ).strip().lower()

        if final_status == "success":

            unassigned_success.append(procedure_name)

        elif final_status == "failure":

            unassigned_failure.append(procedure_name)

    unassigned_procedures = {

        "success": unassigned_success,

        "failure": unassigned_failure,

        "success_count": len(unassigned_success),

        "failure_count": len(unassigned_failure),

        "total": (
            len(unassigned_success)
            + len(unassigned_failure)
        )

    }

    # --------------------------------------------------
    # 9. FINAL RESULT
    # --------------------------------------------------

    return {

        "project_name": project_name,

        "date_range": date_range,

        "date_mode": date_mode,

        "subsystems": results,

        "unassigned_procedures": unassigned_procedures

    }
```

```
UPLOAD_FOLDER = os.path.join(

ALLOWED_PDF = set(["pdf"])
ALLOWED_DOCX = set(["docx"])


app = Flask(__name__)

app.secret_key = "satellite_dashboard_local_key"

# --------------------------------------------------
# 1. CREATE FOLDERS
# --------------------------------------------------

def create_folders():

    for folder in [
        INPUT_DIR,
        OUTPUT_DIR,
        UPLOAD_FOLDER
    ]:

        if not os.path.isdir(folder):
            os.makedirs(folder)


# --------------------------------------------------
# 2. FILE VALIDATION
# --------------------------------------------------

def allowed_file(filename, extensions):

    return (
        "." in filename
        and filename.rsplit(".", 1)[1].lower()
        in extensions
    )


# --------------------------------------------------
# 3. CLEAR PREVIOUS FILES
# --------------------------------------------------

def clear_folder(folder):

    if not os.path.isdir(folder):
        return

    for filename in os.listdir(folder):

        path = os.path.join(folder, filename)

        if os.path.isfile(path):

            os.remove(path)

        elif os.path.isdir(path):

            shutil.rmtree(path)


def clear_previous_files():

    clear_folder(INPUT_DIR)
    clear_folder(OUTPUT_DIR)
    clear_folder(UPLOAD_FOLDER)

    subsystems_path = os.path.join(
        BASE_DIR,
        "subsystems.json"
    )

    if os.path.isfile(subsystems_path):

        os.remove(subsystems_path)


# --------------------------------------------------
# 4. RUN PDF CONVERTER
# --------------------------------------------------

def run_pdf_converter():

    main_path = os.path.join(
        BASE_DIR,
        "main.py"
    )

    result = subprocess.run(

        [sys.executable, main_path],

        cwd=BASE_DIR,

        stdout=subprocess.PIPE,

        stderr=subprocess.PIPE,

        universal_newlines=True
    )

    if result.returncode != 0:

        raise RuntimeError(
            result.stderr
            or result.stdout
            or "PDF conversion failed."
        )

    return result.stdout


# --------------------------------------------------
# 5. GET GENERATED JSON
# --------------------------------------------------

def get_latest_json():

    files = glob.glob(
        os.path.join(
            OUTPUT_DIR,
            "output_*.json"
        )
    )

    if not files:
        return None

    return max(
        files,
        key=os.path.getmtime
    )

        # ------------------------------------------
```

# 13. Important behaviour of the final application

| Requirement                             | Implementation |
| --------------------------------------- | -------------- |
| Multiple PDF uploads                    | Yes            |
| Word document upload                    | Yes            |
| Automatic JSON generation               | Yes            |
| Automatic subsystem JSON generation     | Yes            |
| Filename matching without extensions    | Yes            |
| 90% similarity matching                 | Yes            |
| Duplicate filenames within a subsystem  | Removed        |
| Same filename in different subsystems   | Allowed        |
| Ambiguous matching                      | Handled        |
| Overall date option                     | Yes            |
| Start and end date filtering            | Yes            |
| Latest execution within selected period | Yes            |
| Success and Failure dashboard           | Yes            |
| Unassigned procedures                   | Yes            |
| Replace previous submission             | Yes            |
| Project name displayed                  | Yes            |

A few limitations to be aware of before relying on it:

* The Word parser expects subsystem headings to appear as paragraphs immediately before their corresponding tables, with filenames in the third column. Your actual Word document must follow that arrangement.
