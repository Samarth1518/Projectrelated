from docx import Document
from docx.text.paragraph import Paragraph
from docx.table import Table
from docx.oxml.text.paragraph import CT_P
from docx.oxml.table import CT_Tbl

import re


def extract_filenames(cell_text):
    """
    Extract filenames from a Filename cell.
    Handles commas, semicolons, newlines and split extensions.
    """

    if not cell_text:
        return []

    # Treat commas and semicolons as separators.
    text = cell_text.replace(",", "\n")
    text = text.replace(";", "\n")

    lines = [
        line.strip()
        for line in text.splitlines()
        if line.strip()
    ]

    filenames = []

    for line in lines:

        line = re.sub(r"\s+", "", line)

        if not line:
            continue

        # Reconstruct a filename split across lines.
        if filenames and (
            filenames[-1].endswith("-")
            or filenames[-1].lower().endswith(".t")
            or filenames[-1].lower().endswith(".s")
        ):
            filenames[-1] += line

        else:
            filenames.append(line)

    # Remove empty values and duplicates.
    result = []
    seen = set()

    for filename in filenames:

        filename = filename.strip()

        if not filename:
            continue

        key = filename.lower()

        if key not in seen:
            seen.add(key)
            result.append(filename)

    return result


def normalize_header(value):
    return re.sub(r"[\s.]+", "", value).lower()


def main():

    doc = Document("subsystem.docx")

    subsystem_data = {}
    current_subsystem = None

    for element in doc.element.body.iterchildren():

        # -------------------------
        # READ PARAGRAPHS
        # -------------------------

        if isinstance(element, CT_P):

            paragraph = Paragraph(element, doc)
            text = paragraph.text.strip()

            if text:

                match = re.match(
                    r"^\s*Subsystem\s*:\s*(.*?)\s*$",
                    text,
                    re.IGNORECASE
                )

                if match:

                    current_subsystem = match.group(1).strip()

                    if current_subsystem:
                        subsystem_data.setdefault(
                            current_subsystem, []
                        )

                    print(
                        "\nCURRENT SUBSYSTEM:",
                        current_subsystem
                    )

        # -------------------------
        # READ TABLES
        # -------------------------

        elif isinstance(element, CT_Tbl):

            table = Table(element, doc)

            if not current_subsystem:
                continue

            print("\nTABLE FOUND:", current_subsystem)

            filename_column = None
            header_row = None

            # Find the Filename header.
            for row_index, row in enumerate(table.rows):

                values = [
                    cell.text.strip()
                    for cell in row.cells
                ]

                for column_index, value in enumerate(values):

                    if normalize_header(value) == "filename":
                        filename_column = column_index
                        header_row = row_index
                        break

                if filename_column is not None:
                    break

            if filename_column is None:
                print("WARNING: Filename column not found")
                continue

            print("Filename column:", filename_column + 1)

            # Extract filename values below the header.
            for row in table.rows[header_row + 1:]:

                if filename_column >= len(row.cells):
                    continue

                cell_text = row.cells[filename_column].text

                extracted = extract_filenames(cell_text)

                for filename in extracted:

                    if filename not in subsystem_data[current_subsystem]:
                        subsystem_data[current_subsystem].append(
                            filename
                        )

    print("\n\n========== FINAL SUBSYSTEM DATA ==========")

    for subsystem, filenames in subsystem_data.items():

        print("\nSUBSYSTEM:", subsystem)

        for filename in filenames:
            print("  ", filename)


if __name__ == "__main__":
    main()
```

from docx import Document
from docx.text.paragraph import Paragraph
from docx.table import Table
from docx.oxml.text.paragraph import CT_P
from docx.oxml.table import CT_Tbl

import re
import os


def normalize_header(value):
    return re.sub(r"[\s.]+", "", value).lower()


def is_valid_filename(value):

    value = value.strip()

    if not value:
        return False

    # Reject obvious non-filename entries.
    if value.startswith("(") or value.endswith(")"):
        return False

    if ":" in value:
        return False

    # Reject entries containing ordinary sentence spaces.
    if " " in value:
        return False

    # Accept known procedure/script extensions.
    if re.search(r"\.(tst|sh)$", value, re.IGNORECASE):
        return True

    # Accept extensionless names only when they look like
    # procedure identifiers.
    if re.match(r"^[A-Za-z0-9_-]+$", value):
        return True

    return False


def extract_filenames(cell_text):

    if not cell_text:
        return []

    text = cell_text.replace(",", "\n")
    text = text.replace(";", "\n")

    lines = [
        line.strip()
        for line in text.splitlines()
        if line.strip()
    ]

    reconstructed = []

    for line in lines:

        line = line.strip()

        # Join fragments split across lines.
        if reconstructed and (
            reconstructed[-1].endswith("-")
            or reconstructed[-1].lower().endswith(".t")
            or reconstructed[-1].lower().endswith(".s")
        ):
            reconstructed[-1] += line

        else:
            reconstructed.append(line)

    result = []
    seen = set()

    for filename in reconstructed:

        # Remove whitespace inside split filenames.
        filename = re.sub(r"\s+", "", filename)

        if not is_valid_filename(filename):
            continue

        key = filename.lower()

        if key not in seen:
            seen.add(key)
            result.append(filename)

    return result


def main():

    doc = Document("subsystem.docx")

    subsystem_data = {}

    current_subsystem = None

    for element in doc.element.body.iterchildren():

        # -------------------------
        # READ PARAGRAPHS
        # -------------------------

        if isinstance(element, CT_P):

            paragraph = Paragraph(element, doc)

            text = paragraph.text.strip()

            match = re.match(
                r"^\s*Subsystem\s*:\s*(.*?)\s*$",
                text,
                re.IGNORECASE
            )

            if match:

                current_subsystem = match.group(1).strip()

                if current_subsystem:
                    subsystem_data.setdefault(
                        current_subsystem, []
                    )

                print(
                    "\nCURRENT SUBSYSTEM:",
                    current_subsystem
                )

        # -------------------------
        # READ TABLES
        # -------------------------

        elif isinstance(element, CT_Tbl):

            table = Table(element, doc)

            if not current_subsystem:
                continue

            print("\nTABLE FOUND:", current_subsystem)

            filename_column = None
            header_row = None

            # Find Filename column dynamically.
            for row_index, row in enumerate(table.rows):

                values = [
                    cell.text.strip()
                    for cell in row.cells
                ]

                for column_index, value in enumerate(values):

                    if normalize_header(value) == "filename":

                        filename_column = column_index
                        header_row = row_index
                        break

                if filename_column is not None:
                    break

            if filename_column is None:

                print("WARNING: Filename column not found")
                continue

            print(
                "Filename column:",
                filename_column + 1
            )

            # Extract filenames.
            for row in table.rows[header_row + 1:]:

                if filename_column >= len(row.cells):
                    continue

                cell_text = row.cells[filename_column].text

                extracted = extract_filenames(cell_text)

                for filename in extracted:

                    existing = [
                        item.lower()
                        for item in subsystem_data[current_subsystem]
                    ]

                    if filename.lower() not in existing:

                        subsystem_data[current_subsystem].append(
                            filename
                        )

    # -------------------------
    # DISPLAY FINAL RESULT
    # -------------------------

    print("\n\n========== FINAL SUBSYSTEM DATA ==========")

    for subsystem, filenames in subsystem_data.items():

        print("\nSUBSYSTEM:", subsystem)

        for filename in filenames:
            print("   ", filename)

    print("\nTOTAL SUBSYSTEMS:", len(subsystem_data))

    total_files = sum(
        len(files)
        for files in subsystem_data.values()
    )

    print("TOTAL FILENAMES:", total_files)


if __name__ == "__main__":
    main()
```
