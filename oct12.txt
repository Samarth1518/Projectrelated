## 4. Actual PDF Text Structure

The PDFs contain page-wise extracted text. The inspection reports showed headings and data similar to the following.

### Procedure and page metadata

The report can contain:

* Page number
* Procedure
* Time
* Mode
* Source
* Duration (`Dur`)
* Priority

These values do not necessarily appear together immediately before every table. Some metadata may appear around page boundaries or between tables.

### STEP table heading

The actual heading commonly appears across three lines:

```text
STEP ID    STEP
TIME
STATUS
```

This is important because the original inspection code expected the `TIME` and `STATUS` headings to be on the same line or across only two lines.

### STEP record

A typical record looks like this:

```text
01.004    dsvgsahlasdghlfjkv
11-Dec-2025 11:44:35:234    SUCCESS
```

The STEP ID and description appear on one line. The timestamp and status appear on the following line.

The parser must associate the two lines with the same STEP record.

### Unrelated MODE row

Sometimes a row appears similar to:

```text
MODE    AUTO    11-Dec-2025    11:44:34:256    FAILURE
```

This is an illustrative example of the issue reported from the PDFs.

The existing extraction problem was that the `MODE AUTO` portion could be ignored while the timestamp and `FAILURE` were still incorrectly extracted.

**Required behavior:** if the row is not a genuine STEP record, the entire row must be excluded from STEP extraction. Its timestamp and status must not become a STEP record.

def main():

    print("PDF INSPECTION")
    print("=" * 60)

    if not os.path.isdir(INPUT_DIR):
        print("ERROR: Input folder not found:", INPUT_DIR)
        return

    pdf_files = sorted(
        name for name in os.listdir(INPUT_DIR)
        if name.lower().endswith(".pdf")
    )

    print("PDF files found:", len(pdf_files))

    if not pdf_files:
        print("ERROR: No PDF files found in input folder.")
        return

    report = []

    for filename in pdf_files:
        path = os.path.join(INPUT_DIR, filename)

        report.append("\n" + "=" * 60)
        report.append("PDF: " + filename)
        report.append("=" * 60)

        print("\nProcessing:", filename)

        try:
            with fitz.open(path) as document:
                print("Pages:", len(document))
                report.append(
                    "Pages: " + str(len(document))
                )

                for page_index in range(len(document)):
                    page = document[page_index]
                    text = page.get_text("text")

                    report.append(
                        "\n--- PAGE "
                        + str(page_index + 1)
                        + " ---"
                    )
                    report.append(text)

                    print(
                        " Page", page_index + 1,
                        "- extracted characters:", len(text)
                    )

                print("Finished:", filename)

        except Exception as error:
            message = (
                "ERROR reading "
                + filename + ": " + str(error)
            )
            print(message)
            report.append(message)

    with open(
        REPORT_PATH, "w", encoding="utf-8"
    ) as output:
        output.write("\n".join(report))

    print("\nReport saved to:", REPORT_PATH)
    print("PDF inspection finished.")


if __name__ == "__main__":
    main()

import os
import re

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
SOURCE_PATH = os.path.join(
    BASE_DIR, "inspect_all_pdfs_report.txt"
)
OUTPUT_PATH = os.path.join(
    BASE_DIR, "inspect_step_tables_report.txt"
)

# Recognize headings even when their columns are split
# across two lines.
STEP_ID_HEADER = re.compile(r"\bSTEP\s*ID\b", re.IGNORECASE)
STEP_HEADER = re.compile(r"\bSTEP\b", re.IGNORECASE)
TIME_HEADER = re.compile(r"\bTIME\b", re.IGNORECASE)
STATUS_HEADER = re.compile(r"\bSTATUS\b", re.IGNORECASE)

STEP_ID_VALUE = re.compile(
    r"^\s*(\d{2}\.\d{3,})\s+"
)

def is_table_header(lines, index):
    current = lines[index].strip()

    if (
        STEP_ID_HEADER.search(current)
        and STEP_HEADER.search(current)
        and TIME_HEADER.search(current)
        and STATUS_HEADER.search(current)
    ):
        return True

    if index + 1 >= len(lines):
        return False

    following = lines[index + 1].strip()

    # Supports:
    # STEP ID    STEP
    # TIME       STATUS
    return (
        STEP_ID_HEADER.search(current)
        and STEP_HEADER.search(current)
        and TIME_HEADER.search(following)
        and STATUS_HEADER.search(following)
    )


def main():
    if not os.path.isfile(SOURCE_PATH):
        print("ERROR: Source report not found:")
        print(SOURCE_PATH)
        return

    with open(
        SOURCE_PATH,
        "r",
        encoding="utf-8",
        errors="replace"
    ) as file:
        lines = file.readlines()

    output = [
        "STEP TABLE HEADING INSPECTION",
        "Source: inspect_all_pdfs_report.txt",
        "Total lines: " + str(len(lines)),
        "",
    ]

    header_count = 0
    id_count = 0
    mode_count = 0

    for index, raw_line in enumerate(lines):
        line = raw_line.strip()

        if is_table_header(lines, index):
            header_count += 1
            output.append(
                f"TABLE HEADER {header_count}, line {index + 1}"
            )

            start = max(0, index - 2)
            end = min(len(lines), index + 12)

            for j in range(start, end):
                output.append(
                    f"{j + 1}: {lines[j].rstrip()}"
                )

            output.append("")

        if STEP_ID_VALUE.match(line):
            id_count += 1

        if re.match(r"^MODE\b", line, re.IGNORECASE):
            mode_count += 1

    output.extend([
        "SUMMARY",
        "Possible table headings: " + str(header_count),
        "Lines starting with candidate STEP IDs: " + str(id_count),
        "Lines starting with MODE: " + str(mode_count),
        "",
        "This report identifies candidate patterns only.",
        "It does not certify that candidate IDs are valid records.",
    ])

    with open(
        OUTPUT_PATH, "w", encoding="utf-8"
    ) as file:
        file.write("\n".join(output))

    print("STEP table inspection finished.")
    print("Possible table headings:", header_count)
    print("Candidate STEP ID lines:", id_count)
    print("MODE lines:", mode_count)
    print("Report saved to:", OUTPUT_PATH)


if __name__ == "__main__":
    main()
