# json_to_csv.py
# Convert generated PDF-to-JSON output into a grouped CSV file.
# Compatible with Python 3.6.5.

import os
import glob
import json
import csv
from datetime import datetime


BASE_DIR = os.path.dirname(os.path.abspath(__file__))
OUTPUT_DIR = os.path.join(BASE_DIR, "output")


def get_latest_json():
    """Find the latest generated JSON file."""

    pattern = os.path.join(OUTPUT_DIR, "output_*.json")
    json_files = glob.glob(pattern)

    if not json_files:
        raise FileNotFoundError(
            "No generated JSON files found in output folder."
        )

    return max(json_files, key=os.path.getmtime)


def get_mode(execution):
    """Determine Auto, Manual or Both from recorded STEP IDs."""

    mode_data = execution.get("mode_executions", {})

    auto_data = mode_data.get("auto", {})
    manual_data = mode_data.get("manual", {})

    status_names = [
        "success",
        "failure",
        "progress",
        "suspend"
    ]

    auto_has_steps = False
    manual_has_steps = False

    for status in status_names:

        if auto_data.get(status, []):
            auto_has_steps = True

        if manual_data.get(status, []):
            manual_has_steps = True

    if auto_has_steps and manual_has_steps:
        return "Both (Auto + Manual)"

    elif auto_has_steps:
        return "Auto"

    elif manual_has_steps:
        return "Manual"

    else:
        return "Unknown"


def format_execution_date(execution):
    """
    Extract only the date from the JSON timestamp.

    Input:
    2024-10-01 09:39:36.822

    Output:
    01-Oct-2024
    """

    date_value = execution.get("date_time", "")

    if not date_value:
        return ""

    formats = [
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S",
        "%d-%b-%Y %H:%M:%S:%f",
        "%d-%b-%Y %H:%M:%S"
    ]

    for date_format in formats:
        try:
            parsed_date = datetime.strptime(
                date_value.strip(),
                date_format
            )

            return parsed_date.strftime("%d-%b-%Y")

        except ValueError:
            continue

    # Do not silently display an incorrect date.
    return "INVALID DATE: " + str(date_value)


def get_datetime_sort_key(execution):
    """Sort executions chronologically."""

    date_value = execution.get("date_time", "")

    if not date_value:
        return (1, datetime.max)

    formats = [
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S",
        "%d-%b-%Y %H:%M:%S:%f",
        "%d-%b-%Y %H:%M:%S"
    ]

    for date_format in formats:
        try:
            parsed_date = datetime.strptime(
                date_value.strip(),
                date_format
            )

            return (0, parsed_date)

        except ValueError:
            continue

    return (1, datetime.max)


def get_failure_reasons(execution):
    """
    Extract ALL failure reasons with their corresponding STEP IDs.

    Multiple failure reasons are placed in one CSV cell,
    separated by line breaks.
    """

    final_status = execution.get("final_status", "")

    if str(final_status).strip().lower() != "failure":
        return ""

    reasons = execution.get("final_status_reason", [])

    if not isinstance(reasons, list):
        return ""

    formatted_reasons = []

    for reason_entry in reasons:

        if not isinstance(reason_entry, dict):
            continue

        for step_id, description in reason_entry.items():

            description = str(description).strip()

            if description:
                formatted_reasons.append(
                    str(step_id) + ": " + description
                )
            else:
                formatted_reasons.append(str(step_id))

    return "\n".join(formatted_reasons)


def convert_json_to_csv():

    print("=" * 60)
    print("JSON TO CSV CONVERTER")
    print("=" * 60)

    if not os.path.isdir(OUTPUT_DIR):
        raise RuntimeError("Output folder does not exist.")

    # Find latest JSON file.
    json_path = get_latest_json()

    print("\nReading JSON:", os.path.basename(json_path))

    with open(json_path, "r", encoding="utf-8") as json_file:
        data = json.load(json_file)

    # Generate CSV with the same filename timestamp.
    csv_name = os.path.splitext(
        os.path.basename(json_path)
    )[0] + ".csv"

    csv_path = os.path.join(OUTPUT_DIR, csv_name)

    serial_number = 1
    total_procedures = 0
    total_executions = 0
    total_failures = 0
    total_failure_reasons = 0

    headers = [
        "Si No",
        "Procedure Name",
        "Date of Execution",
        "Duration",
        "Mode",
        "Final Status",
        "Failure Reason"
    ]

    with open(
        csv_path,
        "w",
        newline="",
        encoding="utf-8-sig"
    ) as csv_file:

        writer = csv.writer(csv_file)

        writer.writerow(headers)

        # Process each procedure.
        for procedure_name, procedure_data in data.items():

            if procedure_name == "Insights":
                continue

            if not isinstance(procedure_data, dict):
                continue

            executions = procedure_data.get(
                "procedures", []
            )

            if not isinstance(executions, list):
                continue

            if not executions:
                continue

            # Oldest execution first.
            executions = sorted(
                executions,
                key=get_datetime_sort_key
            )

            first_execution = True

            for execution in executions:

                if not isinstance(execution, dict):
                    continue

                date_time = format_execution_date(execution)

                duration = execution.get("Dur", "")

                mode = get_mode(execution)

                final_status = execution.get(
                    "final_status", ""
                )

                failure_reason = get_failure_reasons(execution)

                if str(final_status).strip().lower() == "failure":
                    total_failures += 1
                    if failure_reason:
                        total_failure_reasons += len(
                            failure_reason.split("\n")
                        )

                if first_execution:

                    writer.writerow([
                        serial_number,
                        procedure_name,
                        date_time,
                        duration,
                        mode,
                        final_status,
                        failure_reason
                    ])

                    first_execution = False
                    serial_number += 1
                    total_procedures += 1

                else:

                    writer.writerow([
                        "",
                        "",
                        date_time,
                        duration,
                        mode,
                        final_status,
                        failure_reason
                    ])

                total_executions += 1

    print("\nConversion completed successfully!")

    print("JSON file:", os.path.basename(json_path))
    print("CSV file:", os.path.basename(csv_path))

    print("Total procedures:", total_procedures)
    print("Total executions:", total_executions)
    print("Failed executions:", total_failures)
    print("Failure reasons extracted:", total_failure_reasons)

    print("\nCSV saved at:")
    print(csv_path)

    print("=" * 60)

        convert_json_to_csv()

    except Exception as error:
        print("\nERROR:", str(error))

            dir=OUTPUT_DIR
        )

        os.close(fd)

        with open(
            temp_path,
            "w",
            newline="",
            encoding="utf-8-sig"
        ) as csv_file:

            writer = csv.writer(csv_file)

            writer.writerow(headers)

            for procedure_name, procedure_data in data.items():

                if procedure_name == "Insights":
                    continue

                if not isinstance(procedure_data, dict):
                    continue

                executions = procedure_data.get(
                    "procedures", []
                )

                if not isinstance(executions, list):
                    continue

                executions = [
                    item for item in executions
                    if isinstance(item, dict)
                ]

                if not executions:
                    continue

                # Oldest execution first.
                executions = sorted(
                    executions,
                    key=get_datetime_sort_key
                )

                first_execution = True

                for execution in executions:

                    # Preserve complete JSON date and time.
                    date_time = execution.get(
                        "date_time", ""
                    )

                    duration = execution.get(
                        "Dur", ""
                    )

                    mode = get_mode(execution)

                    final_status = execution.get(
                        "final_status", ""
                    )

                    failure_reasons = get_failure_reasons(
                        execution
                    )

                    if first_execution:

                        writer.writerow([
                            serial_number,
                            procedure_name,
                            date_time,
                            duration,
                            mode,
                            final_status,
                            failure_reasons
                        ])

                        serial_number += 1
                        total_procedures += 1
                        first_execution = False

                    else:

                        writer.writerow([
                            "",
                            "",
                            date_time,
                            duration,
                            mode,
                            final_status,
                            failure_reasons
                        ])

                    total_executions += 1

        # Validate temporary CSV.
        with open(
            temp_path,
            "r",
            encoding="utf-8-sig",
            newline=""
        ) as check_file:

            reader = csv.reader(check_file)
            rows = list(reader)

            if not rows or rows[0] != headers:
                raise ValueError(
                    "Generated CSV validation failed."
                )

            if len(rows) != total_executions + 1:
                raise ValueError(
                    "CSV execution count validation failed."
                )

        # Remove old CSVs only after the new CSV is ready.
        remove_old_csv_files()

        # Move the validated temporary CSV into its final location.
        os.replace(temp_path, csv_path)
        temp_path = None

        return total_procedures, total_executions

    finally:

        if temp_path and os.path.exists(temp_path):

            try:
                os.remove(temp_path)

            except OSError:
                pass
